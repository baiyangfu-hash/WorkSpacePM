def git(*a: str) -> str:
    return subprocess.run(["git", "-C", str(ROOT), "-c", "core.quotePath=false", *a], capture_output=True, check=True).stdout.decode("utf-8")


def sha(b: bytes) -> str:
    return hashlib.sha256(b).hexdigest()


def sw(t: str, a: str, b: str, n: int = 1) -> str:
    assert t.count(a) == n, (a[:100], t.count(a), n)
    return t.replace(a, b)


def sw_line(t: str, startswith: str, new_line: str) -> str:
    """Replace the single whole line that starts with *startswith*."""
    lines = t.split(NL)
    hit = [i for i, ln in enumerate(lines) if ln.startswith(startswith)]
    assert len(hit) == 1, (startswith[:80], len(hit))
    lines[hit[0]] = new_line
    return NL.join(lines)
