"""Derive the switch card D07p V1.1 (CARD-SW008-M3-D07P V1.1) and its deploy script from V1.0 (m3-d07p, audited, run once on 2026-11-05 06:25 MST and rolled back automatically).

V1.0 installed 1.3.3-dd8b192, switched, and its post-switch check failed on `pm orient SW-2026-008` = ["LIVE_RUN_UNDER_FINISHED_WORK"] (a new interlock of the 7.2 runtime; 6 old Runs in SW-2026-008, 2 in SYS-2026-001);
the script rolled the pointers back (active = 1.3.3-d563293 again).  V1.1 is the same switch, to run only after card SETTLE72 has settled those Runs:
 * new ids: Work/Run/Checkpoint D07P-002, Decision suffix M3D07P02, owner claude:m3d07p2, idempotency keys d07p2-*, evidence folder m3-d07p2; the change request CHG-SCPT-2026-479 of V1.0 is reused;
 * new preconditions: SETTLE72 finished (settle_result.txt), no Run under a finished Work in SW-2026-008 or SYS-2026-001 (the 7.2 runtime's own SQL), V1.0's Run D07P-001 and the REL72-002 Run CANCELLED;
 * new post-switch check: `pm orient SYS-2026-001` conflicts must be empty too;
 * a NEW audit reference is required (switch_audit_ref.txt is NOT copied from V1.0); the user's standing authorization record is copied.
usage: python -B derive_d07p2.py       (writes only under .auto-pm/reports/m3-d07p2/; nothing here is governed)
"""

import hashlib
import re
import shutil
from pathlib import Path

ROOT = Path(r"C:\Users\fubai\Documents\My_Workspace")
R = ROOT / ".auto-pm" / "reports"
SRC, EV = R / "m3-d07p", R / "m3-d07p2"
EV.mkdir(parents=True, exist_ok=True)
NL = chr(10)
RID = "1.3.3-dd8b192"


def sha(b: bytes) -> str:
    return hashlib.sha256(b).hexdigest()


def sw(t: str, a: str, b: str, n: int = 1) -> str:
    assert t.count(a) == n, (a[:110], t.count(a), n)
    return t.replace(a, b)


def ren(t: str) -> str:
    t = t.replace(".auto-pm/worktrees/m3-d07p", "@@WT@@")
    t = t.replace("m3-d07p", "m3-d07p2")
    t = t.replace("@@WT@@", ".auto-pm/worktrees/m3-d07p")
    t = t.replace("D07P-001", "D07P-002").replace("M3D07P01", "M3D07P02").replace("claude:m3d07p", "claude:m3d07p2")
    t = t.replace("d07p-", "d07p2-")
    t = t.replace("CARD-SW008-M3-D07P V1.0", "CARD-SW008-M3-D07P V1.1")
    return t


shutil.copy2(SRC / "d07p_chg.txt", EV / "d07p_chg.txt")  # V1.1 continues the change request of V1.0 (CHG-SCPT-2026-479)
shutil.copy2(SRC / "user_switch_confirmation.txt", EV / "user_switch_confirmation.txt")  # the user's standing authorization record; switch_audit_ref.txt is deliberately NOT copied

s = ren((SRC / "d07p_deploy.sh").read_text(encoding="utf-8"))
s = sw(s, "# M3-D07P (card D07p_card.md V1.0)", "# M3-D07P (card D07p_card.md V1.1)")
# new preconditions (after the worktree-count guard that V1.0 already has)
guard = 'nwt="$(ls -1 "$REAL/.auto-pm/worktrees" | wc -l)"'
i = s.index(guard)
j = s.index(NL, i)
extra = NL.join([
    '[[ "$(cat "$REAL/.auto-pm/reports/m3-settle72/settle_result.txt" 2>/dev/null)" == SETTLE72_DONE* ]] || die "card SETTLE72 has not finished (m3-settle72/settle_result.txt)"',
    'for rr in RUN-SW008-M3-D07P-001 RUN-SW008-M3-REL72-002; do [[ "$(q "select state from run_items where run_id=?" "$rr")" == CANCELLED ]] || die "$rr is not CANCELLED (card SETTLE72)"; done',
    r'''for pj in SW-2026-008 SYS-2026-001; do [[ "$("$PY" -B -c "import sqlite3,sys;c=sqlite3.connect('file:.auto-pm/continuity.db?mode=ro',uri=True);print(c.execute('SELECT count(*) FROM run_items r JOIN work_items w ON w.work_id=r.work_id WHERE w.subject_project_id=? AND r.state NOT IN (?,?,?) AND w.state IN (?,?,?)',(sys.argv[1],'SUCCEEDED','FAILED','CANCELLED','ACCEPTED','CLOSED','CANCELLED')).fetchone()[0])" "$pj")" == 0 ]] || die "$pj still has a Run under a finished Work (the 7.2 interlock would fail the switch again)"; done''',
])
s = s[: j + 1] + extra + NL + s[j + 1:]
# new post-switch check: the same orient conflict check for SYS-2026-001
anchor = 'echo "smoke: pm orient conflicts=$oc"; [[ "$oc" == "[]" ]] || { echo "FAIL smoke orient conflicts"; fail=1; }'
assert s.count(anchor) == 1
s = s.replace(anchor, anchor + NL + '  oc2="$(cli pm orient SYS-2026-001 --json 2>/dev/null | "$PY" -B -c \'import sys,json;t=sys.stdin.read();d=json.JSONDecoder().raw_decode(t[t.index("{"):])[0];print(json.dumps(d.get("conflicts")))\' 2>/dev/null)"; echo "smoke: pm orient SYS-2026-001 conflicts=$oc2"; [[ "$oc2" == "[]" ]] || { echo "FAIL smoke orient conflicts SYS-2026-001"; fail=1; }')
# ---- step M2: V1.0 left the installed release folder (untracked) and its manifest entry (uncommitted) in the control root; the runtime refuses a Run whose owned range has such drift
# (_reject_control_root_drift).  Put both back to the main state with a reversible move (same idea as step M of card 221 V1.1), then start the Run FIRST and install afterwards, as in V1.0.
old_chk = '''  if [[ "$(entry_sha)" == NONE ]]; then
    [[ "$(sha256sum < "$C/deployment_manifest.json" | cut -d' ' -f1)" == "$OLD_MANIFEST_SHA" ]] || die "manifest drifted"
    [[ ! -e "$C/releases/$RID" ]] || die "release slot exists without a manifest entry"
  fi'''
s = sw(s, old_chk, '  bash "$REAL/.auto-pm/reports/m3-d07p2/d07p2_0_move_back_installed.sh" check || die "step M2 check failed (the installed 1.3.3-dd8b192 folder and its manifest entry left by V1.0)"')
s = sw(s, 'echo "== 3. Work and Run on the control root"', 'echo "== 2b. Step M2: put the V1.0 leftovers (installed release folder + manifest entry) back to the main state"\nbash "$REAL/.auto-pm/reports/m3-d07p2/d07p2_0_move_back_installed.sh" move | tee "$EV/stepM2.out"\n\necho "== 3. Work and Run on the control root"')
s = sw(s, 'if [[ "$(entry_sha)" == NONE ]]; then bash "$OPS" build "$ROOT" "$FROZEN" | tee "$EV/install.txt"; fi',
      'if [[ "$(rid_of "$C/active_release.json")" == "$OLD" ]]; then\n  [[ "$(entry_sha)" == NONE && ! -e "$C/releases/$RID" ]] || die "$RID is already present before the install (V1.1 starts from a clean state; step M2 should have removed the V1.0 leftovers)"\n  bash "$OPS" build "$ROOT" "$FROZEN" | tee "$EV/install.txt"\nfi')
m2 = f'''#!/usr/bin/env bash
# M3-D07P V1.1, step M2.  V1.0 installed releases/1.3.3-dd8b192 (untracked) and added its manifest entry (uncommitted) in the control root, then rolled the pointers back.  The runtime refuses to start a Run
# whose owned range has such control-root drift, so BEFORE the Run starts this script puts both back to the main state with a reversible move: the folder is renamed to
# .auto-pm/archive/releases_installed_20261105/ (no byte changes, nothing deleted) and deployment_manifest.json is restored to the bytes of the control-root HEAD (its current bytes are backed up first).
# The release is then installed again after the Run has started (step 4 of the deploy script), exactly as in V1.0.
# usage: d07p2_0_move_back_installed.sh check   read-only: either the clean state already holds, or the leftovers are EXACTLY the V1.0 ones
#        d07p2_0_move_back_installed.sh move    check, then back up, move, restore, verify; undo = stepM2/UNDO_M2.sh
set -euo pipefail
export PATH="/usr/bin:/mingw64/bin:$PATH" PYTHONIOENCODING=utf-8 PYTHONUTF8=1
ROOT=/c/Users/fubai/Documents/My_Workspace; EV="$ROOT/.auto-pm/reports/m3-d07p2"; PY="$ROOT/.venv/Scripts/python.exe"
C=00_Infrastructure/auto_pm; RID={RID}; OLD_MANIFEST_SHA=__OLD_MANIFEST_SHA__

[[ -d "$REL" && ! -L "$REL" ]] || die "the installed folder is missing or not a plain directory while the manifest differs from the frozen bytes"
[[ ! -e "$DEST" ]] || die "the target already exists: $DEST"
git ls-files --error-unmatch -- "$C/releases/$RID" > /dev/null 2>&1 && die "$RID is tracked by Git: not a V1.0 leftover"
"$PY" -B -I - "$MAN" "$REL" <<'EOF' || die "the leftovers are not exactly the V1.0 ones"
import json, os, stat, subprocess, sys
man, rel = sys.argv[1:3]

assert {{k: v for k, v in work.items() if k != "releases"}} == {{k: v for k, v in head.items() if k != "releases"}}, "manifest top level differs"
assert {{k: v for k, v in work["releases"].items() if k != rid}} == head["releases"], "other manifest entries differ"
for dp, dn, fn in os.walk(rel):
    for x in dn + fn:
        s2 = os.lstat(os.path.join(dp, x))
        assert not stat.S_ISLNK(s2.st_mode) and not (getattr(s2, "st_file_attributes", 0) & 0x400), ("link or junction inside", x)
print("leftovers are exactly the V1.0 ones: the folder", rid, "and its one manifest entry; no links or junctions")
EOF
HEADSHA="$(git show "HEAD:$C/deployment_manifest.json" | sha256sum | cut -d' ' -f1)"
[[ "$HEADSHA" == "$OLD_MANIFEST_SHA" ]] || die "the HEAD manifest bytes differ from the frozen value"
[[ "$MODE" == move ]] || {{ echo "STEP_M2_CHECK_OK: nothing was written"; exit 0; }}
OUT="$EV/stepM2"; mkdir -p "$OUT"
cp "$MAN" "$OUT/deployment_manifest.json.before"
find "$REL" -type f | wc -l > "$OUT/file_count_before.txt"
{{ echo '#!/usr/bin/env bash'; echo '# undo of step M2: puts the installed folder and the V1.0 manifest bytes back'; echo 'set -euo pipefail'; printf 'mv -- "%s" "%s"\\n' "$DEST/$RID" "$REL"; printf 'cp -- "%s" "%s"\\n' "$OUT/deployment_manifest.json.before" "$MAN"; }} > "$OUT/UNDO_M2.sh"
mkdir -p "$DEST"
mv -- "$REL" "$DEST/$RID"
[[ "$(find "$DEST/$RID" -type f | wc -l)" == "$(cat "$OUT/file_count_before.txt")" ]] || die "file count differs after the move (UNDO_M2.sh puts everything back)"
git show "HEAD:$C/deployment_manifest.json" > "$MAN"
[[ "$(sha256sum < "$MAN" | cut -d' ' -f1)" == "$OLD_MANIFEST_SHA" ]] || die "the restored manifest does not equal the frozen bytes (UNDO_M2.sh puts everything back)"
[[ -z "$(git status --porcelain -- "$C/deployment_manifest.json" "$C/releases")" ]] || die "the release files are still dirty after step M2: $(git status --porcelain -- "$C/deployment_manifest.json" "$C/releases" | head -3)"
echo "STEP_M2_OK: the installed $RID folder moved to $DEST, the manifest restored to the HEAD bytes, release files clean"
'''
