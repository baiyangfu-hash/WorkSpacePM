"""Non-governed rehearsal of the three V1.1 judge checks (reuse, pin, sleep) on a temporary gates folder: one positive case and the negative variants.  usage: python -B v11_checks_test.py"""
import importlib.util
import os
import shutil
import sys
import tempfile
import time

ROOT = "C:/Users/fubai/Documents/My_Workspace"
V10 = ROOT + "/.auto-pm/reports/m3-rel71/gates"
tmp = tempfile.mkdtemp(prefix="v11chk_")
os.environ["PG_G"] = tmp
os.environ.pop("PG_REHEARSAL", None)
spec = importlib.util.spec_from_file_location("g", ROOT + "/.auto-pm/reports/m3-rel71b/rel71_gates.py")
g = importlib.util.module_from_spec(spec)
spec.loader.exec_module(g)
res = []


def fresh():
    shutil.rmtree(tmp + "/g2_candidate", ignore_errors=True)
    os.makedirs(tmp + "/g2_candidate")
    for n in g.REUSE:
        shutil.copy(f"{V10}/g2_candidate/{n}", f"{tmp}/g2_candidate/{n}")
    now = time.time()
    for n in ("core.txt", "core.exit", "core.started", "core.finished", "ALL_DONE"):
        open(f"{tmp}/g2_candidate/{n}", "w").write("x\n")
    os.utime(f"{tmp}/g2_candidate/core.started", (now - 3600, now - 3600))
    os.utime(f"{tmp}/g2_candidate/core.finished", (now - 60, now - 60))
    open(tmp + "/pin_start.txt", "w").write(f"HEAD={g.BASE}\nORIGIN_MAIN={g.BASE}\nMAIN_PRODUCT_DIFF=0\n")
    shutil.copy(tmp + "/pin_start.txt", tmp + "/pin_end.txt")
    open(tmp + "/heartbeat.log", "w").write("".join(f"{int(now - 3700 + 30 * i)} t\n" for i in range(130)))
    for n in ("power_start.txt", "power_end.txt"):
        open(f"{tmp}/{n}", "w").write("lid\n")


def case(name, mut, fn, want):
    fresh()
    mut()
    ok, line = fn()
    res.append((name, ok == want, ok, line))


case("reuse positive", lambda: None, g.reuse_check, True)
case("reuse: one output missing", lambda: os.remove(tmp + "/g2_candidate/cli.txt"), g.reuse_check, False)
case("reuse: one extra file", lambda: open(tmp + "/g2_candidate/surplus.txt", "w").write("x"), g.reuse_check, False)
case("reuse: one byte changed", lambda: open(tmp + "/g2_candidate/plc.txt", "ab").write(b"x"), g.reuse_check, False)
case("pin positive", lambda: None, g.pin_check, True)
case("pin: HEAD differs at the end", lambda: open(tmp + "/pin_end.txt", "w").write(f"HEAD={'0' * 40}\nORIGIN_MAIN={g.BASE}\n"), g.pin_check, False)
case("pin: origin/main moved during the run", lambda: open(tmp + "/pin_end.txt", "w").write(f"HEAD={g.BASE}\nORIGIN_MAIN={'1' * 40}\n"), g.pin_check, False)
case("pin: end file missing", lambda: os.remove(tmp + "/pin_end.txt"), g.pin_check, False)
case("sleep positive", lambda: None, g.sleep_check, True)
case("sleep: a 400 s gap in the heartbeat", lambda: open(tmp + "/heartbeat.log", "w").write("".join(f"{int(time.time() - 3700 + 30 * i + (400 if i > 60 else 0))} t\n" for i in range(110))), g.sleep_check, False)
case("sleep: heartbeat does not cover the core run", lambda: open(tmp + "/heartbeat.log", "w").write("".join(f"{int(time.time() - 600 + 30 * i)} t\n" for i in range(18))), g.sleep_check, False)
case("sleep: power evidence missing", lambda: os.remove(tmp + "/power_end.txt"), g.sleep_check, False)
case("sleep: no heartbeat at all", lambda: os.remove(tmp + "/heartbeat.log"), g.sleep_check, False)
# V1.0 FAIL record protected: check against a changed expectation without touching the real files
orig = dict(g.V10_RECORDS)
g.V10_RECORDS[next(iter(g.V10_RECORDS))] = "0" * 64
case("reuse: V1.0 FAIL record sha differs", lambda: None, g.reuse_check, False)
g.V10_RECORDS.clear()
g.V10_RECORDS.update(orig)
out = [f"{'PASS' if r else 'FAIL'}  {n} (got {got}): {line}" for n, r, got, line in res]
print("\n".join(out))
shutil.rmtree(tmp, ignore_errors=True)
sys.exit(0 if all(r for _, r, _, _ in res) else 1)
