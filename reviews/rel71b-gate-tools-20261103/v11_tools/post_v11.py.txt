def rd(p):
    return io.open(p, encoding="utf-8", newline="").read()


def wr(p, t):
    io.open(p, "w", encoding="utf-8", newline="").write(t)


def sw(t, a, b, n=1):
    assert t.count(a) == n, (a[:80], t.count(a))
    return t.replace(a, b)


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()

def reuse_check() -> tuple[bool, str]:
    """Nine reused outputs: exact path set and sha256 (missing, extra or changed file = FAIL); the V1.0 FAIL record is unchanged."""
    if REHEARSAL:
        return True, "reuse check: SKIPPED (rehearsal with synthetic outputs)"

def pin_check() -> tuple[bool, str]:
    """The tested commit is pinned: worktree HEAD == BASE before and after, origin/main did not move during the run (audit requirement 1)."""
    if REHEARSAL:
        return True, "pin check: SKIPPED (rehearsal)"

def sleep_check() -> tuple[bool, str]:
    """No gap above HB_GAP_MAX in the 30 s heartbeat from before the core group started to after it finished; the power-settings evidence files exist (audit requirement 4)."""

    worst = max(gaps) if gaps else 1e9
    cover = bool(ts) and ts[0] <= t0 and ts[-1] >= t1
    ev = all(os.path.exists(f"{G}/{n}") and os.path.getsize(f"{G}/{n}") > 0 for n in ("power_start.txt", "power_end.txt"))
    ok = len(ts) > 2 and worst <= HB_GAP_MAX and cover and ev
    return ok, f"sleep check: {len(ts)} heartbeat samples, largest gap {worst:.0f}s (limit {HB_GAP_MAX}s), covers core run {cover}, power evidence files {ev} -> {'OK' if ok else 'FAIL'}"
