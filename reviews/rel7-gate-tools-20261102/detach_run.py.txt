"""Start a command completely detached from the calling session (no console, own process group, outside the caller's job object when allowed).
usage: python -B detach_run.py <logfile> <cmd> [args...]   -> prints the child's PID; the child keeps running when the session/shell that started it exits."""
import subprocess
import sys

DETACHED_PROCESS = 0x00000008
CREATE_NEW_PROCESS_GROUP = 0x00000200
CREATE_BREAKAWAY_FROM_JOB = 0x01000000
CREATE_NO_WINDOW = 0x08000000

log = open(sys.argv[1], "ab", buffering=0)
cmd = sys.argv[2:]
last = None
for flags in (DETACHED_PROCESS | CREATE_NEW_PROCESS_GROUP | CREATE_BREAKAWAY_FROM_JOB, DETACHED_PROCESS | CREATE_NEW_PROCESS_GROUP):
    try:
        p = subprocess.Popen(cmd, stdin=subprocess.DEVNULL, stdout=log, stderr=subprocess.STDOUT, creationflags=flags, close_fds=True)
        print(p.pid, "flags=0x%x" % flags)
        break
    except OSError as error:
        last = error
else:
    raise SystemExit("cannot start detached: %s" % last)
