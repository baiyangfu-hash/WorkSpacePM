"""Write one stage status file for a detached gate run: <statusdir>/<stage>.json and a line in <statusdir>/status.log.
usage: python -B gate_status.py <statusdir> <stage> <RUNNING|PASS|FAIL|SKIPPED> [exit_code]
Fields: stage, state, exit_code, time_mst (UTC-7, fixed), committed_gb (system commit charge)."""
import csv
import datetime
import json
import subprocess
import sys
from pathlib import Path

SEP = chr(92)


def committed_gb() -> float | None:
    try:
        out = subprocess.run(["typeperf", SEP + "Memory" + SEP + "Committed Bytes", "-sc", "1"], capture_output=True, timeout=30).stdout.decode("gbk", errors="replace")
        rows = [r for r in csv.reader(out.splitlines()) if len(r) > 1]
        return round(float(rows[1][1]) / 2**30, 1)
    except Exception:  # noqa: BLE001 - a status writer must never stop the gates
        return None


def main() -> None:
    d, stage, state = Path(sys.argv[1]), sys.argv[2], sys.argv[3]
    code = int(sys.argv[4]) if len(sys.argv) > 4 else None
    d.mkdir(parents=True, exist_ok=True)
    now = datetime.datetime.now(datetime.timezone.utc)
    rec = {"stage": stage, "state": state, "exit_code": code, "time_mst": (now - datetime.timedelta(hours=7)).strftime("%Y-%m-%d %H:%M:%S MST"),
           "time_utc": now.strftime("%Y-%m-%dT%H:%M:%SZ"), "committed_gb": committed_gb()}
    (d / f"{stage}.json").write_text(json.dumps(rec, ensure_ascii=False) + "\n", encoding="utf-8")
    with open(d / "status.log", "a", encoding="utf-8") as f:
        f.write(json.dumps(rec, ensure_ascii=False) + "\n")


main()
