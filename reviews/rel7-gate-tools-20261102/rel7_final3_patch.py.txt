"""Release 7, audit 558 round 4: approval_refs.env is PARSED line by line (never sourced) with format checks; the close script's receipt cites GO_REF and NIGHT_REF too;
the launcher compares approve_start with the reviewed branch copy (out-of-band anchor) before it starts anything.  Applied ONCE to derive_rel7.py."""
import io

p = "m3-rel6/derive_rel7.py"
s = io.open(p, encoding="utf-8", newline="").read()
if "PARSER_TEXT" in s:
    raise SystemExit("final3 patch already applied: refusing")
NL = chr(10)

# 1) the parser text, defined once in derive (used by the start script and the close script)
anchor = '_st = io.open(EV / "rel7_1_approve_start.sh", encoding="utf-8", newline="").read()' + NL
assert s.count(anchor) == 1
parser_def = r'''
PARSER_TEXT = r"""die() { printf 'ERROR: %s\n' "$*" >&2; exit 1; }
# approval_refs.env is PARSED line by line (never sourced, never eval'd); the refs and the SHA are checked by regular expressions, the text fields reject quotes, backticks, $ and backslashes
load_refs() {
  local f="$1" line key val
  GO_SHA=""; GO_REF=""; GO_TEXT=""; NIGHT_REF=""; NIGHT_TEXT=""
  [[ -f "$f" ]] || die "approval_refs.env is missing: it records the user's own go-ahead for THIS card SHA and is written only after the user said it"
  while IFS= read -r line || [[ -n "$line" ]]; do
    line="${line%$'\r'}"
    [[ -z "$line" || "$line" == \#* ]] && continue
    [[ "$line" == *=* ]] || die "approval_refs.env: a line without '='"
    key="${line%%=*}"; val="${line#*=}"
    case "$key" in
      GO_SHA)  [[ "$val" =~ ^[0-9a-f]{64}$ ]] || die "approval_refs.env: GO_SHA format"; GO_SHA="$val" ;;
      GO_REF)  [[ "$val" =~ ^user-confirmation:cmsg_[A-Za-z0-9]{20,80}$ ]] || die "approval_refs.env: GO_REF format"; GO_REF="$val" ;;
      NIGHT_REF) [[ "$val" =~ ^user-confirmation:cmsg_[A-Za-z0-9]{20,80}$ ]] || die "approval_refs.env: NIGHT_REF format"; NIGHT_REF="$val" ;;
      GO_TEXT|NIGHT_TEXT)
        [[ ${#val} -ge 1 && ${#val} -le 200 && "$val" != *\"* && "$val" != *\'* && "$val" != *\`* && "$val" != *'$'* && "$val" != *\\* ]] || die "approval_refs.env: $key has a rejected character or length"
        printf -v "$key" '%s' "$val" ;;
      *) die "approval_refs.env: unknown key" ;;
    esac
  done < "$f"
  [[ -n "$GO_SHA" && -n "$GO_REF" && -n "$GO_TEXT" ]] || die "approval_refs.env is incomplete"
}
"""
'''
s = s.replace(anchor, parser_def.lstrip(NL) + anchor)

# 2) start script: replace the sourcing block by the parser and the SHA equality check
a = s.index("    '[[ -f \"$EV/approval_refs.env\" ]] || die")
b = s.index("    + 'DELEG=\"委托执行：用户 2026-10-18 预授权自行收口推送；用户 2026-11-02 当地时间")
new_lines = ("    PARSER_TEXT + NL" + NL
             + "    + 'load_refs \"$EV/approval_refs.env\"' + NL" + NL
             + "    + '[[ \"$GO_SHA\" == \"$CARD_SHA\" ]] || die \"approval_refs.env: GO_SHA is not this card SHA\"' + NL" + NL)
s = s[:a] + new_lines + s[b:]

# 3) close script: parse the refs and cite them in the receipt (replaces the 12:04 DELEG)
anchor3 = '_cl = _cl.replace(\'cd "$ROOT"\' + NL'
assert s.count(anchor3) == 1
code3 = r'''
_ca = _cl.index('DELEG="委托执行')
_cb = _cl.index(NL, _ca)
_cl = (_cl[:_ca] + PARSER_TEXT + NL + 'load_refs /c/Users/fubai/Documents/My_Workspace/.auto-pm/reports/m3-rel7/approval_refs.env' + NL
       + 'DELEG="委托执行：用户 2026-10-18 预授权自行收口推送；用户 2026-11-02 当地时间“${NIGHT_TEXT}”（${NIGHT_REF}）并对卡 SHA ${GO_SHA} 亲口说“${GO_TEXT}”（${GO_REF}）；只构建与门禁，不切换有效发布版本（切换要用户另回“切换”）；批准条件：卡字节等于 SHA、范围不超出卡列路径、任一门禁失败即停下保留现场；由 Claude 代为运行脚本；变更单收口在用户阅读结果并确认验收之后"'
       + _cl[_cb:])
'''
s = s.replace(anchor3, code3.lstrip(NL) + anchor3)

# 4) launcher: out-of-band anchor before the start script
old = 'bash "$EV/rel7_1_approve_start.sh" > "$EV/start2.out"'
assert s.count(old) == 1
anchor4 = r'''# out-of-band anchor: approve_start in this folder must equal the copy on the reviewed branch (its hash is also reported in the thread); the comparison goes to launch_anchor.log
git fetch -q origin claude/rel7-gate-tools-20261102 || { echo "ANCHOR: cannot fetch the reviewed branch"; exit 1; }
BRANCH_SHA="$(git show origin/claude/rel7-gate-tools-20261102:reviews/rel7-gate-tools-20261102/rel7_1_approve_start.sh | sha256sum | cut -d' ' -f1)"
LOCAL_SHA="$(sha256sum < "$EV/rel7_1_approve_start.sh" | cut -d' ' -f1)"
echo "$(date -u +%FT%TZ) approve_start anchor: branch=$BRANCH_SHA local=$LOCAL_SHA" | tee -a "$EV/launch_anchor.log"
[ "$BRANCH_SHA" = "$LOCAL_SHA" ] || { echo "ANCHOR MISMATCH: refusing to start"; exit 1; }
'''
s = s.replace(old, anchor4 + old)
io.open(p, "w", encoding="utf-8", newline="").write(s)
print("final3 patch applied")
