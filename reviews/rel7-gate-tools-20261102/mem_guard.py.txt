"""Memory guard for a detached gate run: when the system commit charge reaches the limit it writes ALERT_<time>.txt and SUSPENDS the process tree whose root PID is in <pidfile>;
it resumes the tree when the charge falls below the resume level or a RESUME file appears. Exits when DONE or FAILED appears in <statusdir> or in its parent DIRECTORY (or after 8 h); it does not watch the gates process itself, and if the gates process dies without writing either file it keeps watching until the 8 h limit.
usage: python -B mem_guard.py <statusdir> <pidfile> [limit_gb=40] [resume_gb=36]"""
import csv
import ctypes
import datetime
import subprocess
import sys
import time
from ctypes import wintypes
from pathlib import Path

SEP = chr(92)
k = ctypes.WinDLL("kernel32", use_last_error=True)
nt = ctypes.WinDLL("ntdll")
PROCESS_SUSPEND_RESUME = 0x0800


class PE(ctypes.Structure):
    _fields_ = [("dwSize", wintypes.DWORD), ("cntUsage", wintypes.DWORD), ("th32ProcessID", wintypes.DWORD), ("th32DefaultHeapID", ctypes.c_void_p),
                ("th32ModuleID", wintypes.DWORD), ("cntThreads", wintypes.DWORD), ("th32ParentProcessID", wintypes.DWORD), ("pcPriClassBase", ctypes.c_long),
                ("dwFlags", wintypes.DWORD), ("szExeFile", ctypes.c_wchar * 260)]


def tree(root: int) -> list[int]:
    snap = k.CreateToolhelp32Snapshot(2, 0)
    pe = PE(); pe.dwSize = ctypes.sizeof(PE)
    kids: dict[int, list[int]] = {}
    ok = k.Process32FirstW(snap, ctypes.byref(pe))
    while ok:
        kids.setdefault(pe.th32ParentProcessID, []).append(pe.th32ProcessID)
        ok = k.Process32NextW(snap, ctypes.byref(pe))
    k.CloseHandle(snap)
    out, todo = [], [root]
    while todo:
        p = todo.pop()
        out.append(p)
        todo.extend(kids.get(p, []))
    return out


def act(pids: list[int], fn: str) -> int:
    n = 0
    for pid in pids:
        h = k.OpenProcess(PROCESS_SUSPEND_RESUME, False, pid)
        if h:
            getattr(nt, fn)(h)
            k.CloseHandle(h)
            n += 1
    return n


def committed_gb() -> float:
    out = subprocess.run(["typeperf", SEP + "Memory" + SEP + "Committed Bytes", "-sc", "1"], capture_output=True, timeout=30).stdout.decode("gbk", errors="replace")
    rows = [r for r in csv.reader(out.splitlines()) if len(r) > 1]
    return float(rows[1][1]) / 2**30


def now_mst() -> str:
    return (datetime.datetime.now(datetime.timezone.utc) - datetime.timedelta(hours=7)).strftime("%Y-%m-%d %H:%M:%S MST")


def main() -> None:
    sd, pidfile = Path(sys.argv[1]), Path(sys.argv[2])
    limit = float(sys.argv[3]) if len(sys.argv) > 3 else 40.0
    resume = float(sys.argv[4]) if len(sys.argv) > 4 else 36.0
    sd.mkdir(parents=True, exist_ok=True)
    paused: list[int] = []
    t0 = time.time()
    while time.time() - t0 < 8 * 3600:
        if any((d / n).exists() for d in (sd, sd.parent) for n in ("DONE", "FAILED")):
            break
        try:
            c = committed_gb()
        except Exception:  # noqa: BLE001
            time.sleep(20)
            continue
        if not paused and c >= limit and pidfile.exists():
            try:
                root = int(pidfile.read_text().strip())
            except ValueError:
                root = 0
            if root:
                paused = tree(root)
                n = act(paused, "NtSuspendProcess")
                (sd / f"ALERT_{int(time.time())}.txt").write_text(f"{now_mst()} committed {c:.1f} GB >= {limit} GB: suspended {n} processes of the tree of pid {root}\n", encoding="utf-8")
                (sd / "PAUSED").write_text(f"{now_mst()} committed {c:.1f} GB\n", encoding="utf-8")
        elif paused and (c < resume or (sd / "RESUME").exists()):
            n = act(paused, "NtResumeProcess")
            (sd / "PAUSED").unlink(missing_ok=True)
            (sd / "RESUME").unlink(missing_ok=True)
            (sd / f"RESUMED_{int(time.time())}.txt").write_text(f"{now_mst()} committed {c:.1f} GB: resumed {n} processes\n", encoding="utf-8")
            paused = []
        time.sleep(20)
    if paused:
        act(paused, "NtResumeProcess")


main()
