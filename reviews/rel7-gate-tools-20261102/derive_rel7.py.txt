def orient_extract(parsed: object) -> dict:
    # the run, work id and work state that orient itself reports; anything that is not exactly a dict with string fields is None (never raises)
    out: dict = {"run_id": None, "work_id": None, "work_state": None}
    if not isinstance(parsed, dict):
        return out
    run, work = parsed.get("run"), parsed.get("work")
    if isinstance(run, dict) and isinstance(run.get("run_id"), str):
        out["run_id"] = run["run_id"]
    if isinstance(work, dict):
        out["work_id"] = work["work_id"] if isinstance(work.get("work_id"), str) else None
        out["work_state"] = work["state"] if isinstance(work.get("state"), str) else None
