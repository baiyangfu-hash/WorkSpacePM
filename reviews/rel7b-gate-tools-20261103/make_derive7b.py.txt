s = (R / "m3-rel6" / "derive_rel7.py").read_text(encoding="utf-8")
NL = chr(10)


def rep(t, a, b, n=1):
    assert t.count(a) == n, (a[:80], t.count(a))
    return t.replace(a, b)


# 1) the delegation sentence of the spec
old = "用户 2026-11-02 12:04（当地时间）说“今天电脑给你用，允许白天跑门禁下午5点之前尽量结束”（协调者转述原话，用户消息 cmsg_01AfNxhPWkgwWYuL1EVfYC7tDgoX2UY8rPK5n7nMxruFa2）；"

s = s.replace("m3-rel7", "m3-rel7b")

s = rep(s, "实施卡 `CARD-SW008-M3-REL7 V1.0`", "实施卡 `CARD-SW008-M3-REL7 V1.1`")

        '("M3-D045 V1.1", "M3-REL7 V1.1"), ("M3-D045 (card 171 V1.1)", "M3-REL7 (card 219 V1.1)"), ("card 171 V1.1", "card 219 V1.1")')
s = rep(s, '.replace("Card 182 V1.0", "Card 218 V1.0")', '.replace("Card 182 V1.0", "Card 219 V1.1")')
s = rep(s, "# Card 219 V1.0 launcher (DETACHED)", "# Card 219 V1.1 launcher (DETACHED)")
s = rep(s, '"card_no": "219", "card_id": "CARD-SW008-M3-REL7", "version": "V1.0",', '"card_no": "219", "card_id": "CARD-SW008-M3-REL7", "version": "V1.1",')
s = rep(s, '"card_file": "219_M3-REL7_第七次发版候选门禁实施卡_PM.md"', '"card_file": "219_M3-REL7_第七次发版候选门禁实施卡_V1.1_PM.md"')

s = rep(s, '"work": "WORK-SW008-M3-REL7-001", "run": "RUN-SW008-M3-REL7-001", "ckpt": "CKPT-SW008-M3-REL7-001", "owner": "claude:m3rel7", "branch": "claude/m3-rel7b",',
        '"work": "WORK-SW008-M3-REL7-002", "run": "RUN-SW008-M3-REL7-002", "ckpt": "CKPT-SW008-M3-REL7-002", "owner": "claude:m3rel7b", "branch": "claude/m3-rel7b",')
