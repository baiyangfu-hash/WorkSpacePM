def orient_extract(parsed: object) -> dict:
    # the run, work id and work state that orient itself reports; anything that is not exactly a dict with string fields is None (never raises)
    out: dict = {"run_id": None, "work_id": None, "work_state": None}
    if not isinstance(parsed, dict):
        return out
    run, work = parsed.get("run"), parsed.get("work")
    if isinstance(run, dict) and isinstance(run.get("run_id"), str):
        out["run_id"] = run["run_id"]
    if isinstance(work, dict):
        out["work_id"] = work["work_id"] if isinstance(work.get("work_id"), str) else None
        out["work_state"] = work["state"] if isinstance(work.get("state"), str) else None
    return out

    # (from the same orient JSON) is the known Work and still IN_PROGRESS, and only for this card; `ledger_expired` (other expired Runs in the ledger) is informational and ignored
    if conflicts == []:
        return True
    return (type(conflicts) is list and conflicts == ["LEASE_EXPIRED"] and type(orient_run) is str and orient_run == KNOWN_STUCK_RUN
            and type(work_id) is str and work_id == KNOWN_STUCK_WORK and work_state == "IN_PROGRESS" and card_id == KNOWN_CARD)

def sha(b: bytes) -> str:
    return hashlib.sha256(b).hexdigest()


def sha_file(p: str) -> str:
    return sha(open(p, "rb").read())


def write(name: str, text: str) -> str:

    open(path, "w", encoding="utf-8", newline="\n").write(text)

def sh(args: list[str], cwd: str | None = None, env: dict | None = None, timeout: int = 3600) -> subprocess.CompletedProcess[str]:
    return subprocess.run(args, cwd=cwd, env=env or ENV, capture_output=True, text=True, encoding="utf-8", errors="replace", timeout=timeout)


def git(*args: str, cwd: str | None = None, env: dict | None = None) -> subprocess.CompletedProcess[str]:
    return sh(["git", "-c", "core.quotePath=false", *args], cwd=cwd or WT, env=env)


def rd(p: str) -> str:
    return ANSI.sub("", open(p, encoding="utf-8", errors="replace").read())


def summary(path: str) -> dict:
    lines = [x for x in rd(path).splitlines() if x.strip()]
    last = lines[-1] if lines else ""
    out: dict = {k: int(m.group(1)) if (m := re.search(rf"(\d+) {k}", last)) else 0 for k in ("passed", "failed", "skipped", "error", "deselected")}
    out["last"] = last
    out["failed_ids"] = sorted(x[len("FAILED "):].split(" - ")[0] for x in lines if x.startswith("FAILED "))
    out["error_nodes"] = [x for x in lines if x.startswith("ERROR ")]
    out["exit"] = int(rd(path[:-4] + ".exit").strip())
    return out


def pytest_run(name: str, *targets: str) -> dict:
    done = sh([PY, "-B", "-m", "pytest", "-o", "addopts=", "--no-cov", "-p", "no:cacheprovider", "--color=no", "-q", *targets], cwd=PROJ)
    open(f"{G}/{name}.txt", "w", encoding="utf-8").write(done.stdout + done.stderr)
    open(f"{G}/{name}.exit", "w").write(str(done.returncode) + "\n")
    return summary(f"{G}/{name}.txt")

def tree_digest(base: Path) -> dict[str, str]:
    return {p.relative_to(base).as_posix(): sha_file(str(p)) for p in sorted(base.rglob("*")) if p.is_file() and "__pycache__" not in p.parts and p.suffix != ".pyc"}

SECRET = re.compile(r"-----BEGIN (?:RSA |OPENSSH |EC |DSA |PGP )?PRIVATE KEY|AKIA[0-9A-Z]{16}|(?:ghp_|gho_|ghu_|ghs_|github_pat_)[A-Za-z0-9_]{20,}|xox[baprs]-[A-Za-z0-9-]{10,}|\bsk-[A-Za-z0-9]{32,}")


def docs_contract(texts: dict[str, bytes]) -> dict:
    """Each written-back document: LF only, no BOM, no control character, ends with one newline, no secret-like value; the deleted base lines are exactly the frozen set
    (by SHA-256), every other base line stays in order, and the added lines carry the required tokens."""
    report: dict = {}
    for path, spec in CFG["docs"].items():
        data = texts[path]
        base = subprocess.run(["git", "-c", "core.quotePath=false", "-C", str(ROOT), "show", f"{BASE}:{path}"], capture_output=True).stdout.decode("utf-8")
        out: dict = {"lf_only": b"\r" not in data, "no_bom": not data.startswith(b"\xef\xbb\xbf"), "no_control": not re.search(rb"[\x00-\x08\x0b\x0c\x0e-\x1f]", data),
                     "ends_newline": data.endswith(b"\n") and not data.endswith(b"\n\n")}
        text = data.decode("utf-8", errors="replace")
        old_lines, new_lines = base.split(NL), text.split(NL)
        new_set, old_set = set(new_lines), set(old_lines)
        deleted = sorted(sha(ln.encode("utf-8")) for ln in old_lines if ln not in new_set and ln.strip())
        out["deleted_as_frozen"] = deleted == sorted(spec["deleted_sha256"])
        kept = [ln for ln in old_lines if ln in new_set]
        it = iter(new_lines)
        out["kept_in_order"] = bool(spec.get("rewrite")) or all(any(ln == c for c in it) for ln in kept)
        added = NL.join(ln for ln in new_lines if ln not in old_set)
        out["required_tokens_missing"] = [t for t in spec["required_tokens"] if t not in added]
        out["secret_like"] = bool(SECRET.search(text))
        out["ok"] = bool(out["lf_only"] and out["no_bom"] and out["no_control"] and out["ends_newline"] and out["deleted_as_frozen"] and out["kept_in_order"]
                         and not out["required_tokens_missing"] and not out["secret_like"])
        report[path] = out
    return report

def stage_baseline() -> int:
    head = git("rev-parse", "HEAD").stdout.strip()
    status = git("status", "--porcelain=v1", "--untracked-files=all").stdout
    states = {}
    for r, meta in FILES.items():
        p = Path(WT) / r
        if meta["base_sha256"] is None:
            states[r] = {"expect": "absent", "ok": not p.exists()}
        else:
            states[r] = {"expect": meta["base_sha256"], "ok": p.is_file() and sha_file(str(p)) == meta["base_sha256"]}
    write("baseline_identity.json", json.dumps({"head": head, "head_expected": BASE, "status": status, "files": states}, ensure_ascii=False, indent=1) + "\n")
    write("real_hooks_before.txt", hooks_record())
    write("python_procs_before.txt", py_procs() + "\n")
    write("status_baseline.txt", status)
    binary = plugin_binary()
    write("plugin_identity.json", json.dumps({"version": PLUGIN["version"], "binary": str(binary), "sha256": sha_file(str(binary)), "size": binary.stat().st_size}, indent=1) + "\n")
    pytest_run("G1_baseline_candidate", "tests/test_fixture_health.py", "tests/test_smoke.py", "tests/test_import.py")
    doc_check("G3_doc_check_baseline")
    write("BASELINE_DONE", "ok\n")
    return 0 if head == BASE and not status.strip() and all(s["ok"] for s in states.values()) else 1

        ("D1-the-report-loses-its-proof-sentence", report, lambda t: t.replace("本报告的结论由提交本身证明", "本报告的结论见证据")),
        ("D2-the-report-loses-the-frozen-commit", report, lambda t: t.replace(CFG["frozen_commit"], CFG["frozen_commit"][:7])),
    ):
        variant = dict(texts)
        variant[path] = mutate(texts[path].decode("utf-8")).encode("utf-8")
        assert variant[path] != texts[path], mid
        report = docs_contract(variant)
        mutations.append({"id": mid, "file": path, "killed": not report[path]["ok"], "must_fail": ["docs_contract"], "failed": [k for k, v in report[path].items() if v is False][:4],
                          "restored_bytes_equal": (Path(WT) / path).read_bytes() == texts[path], "last": ""})
    tampered = Path(G) / "d04" / "build_a_tampered"
    shutil.rmtree(tampered, ignore_errors=True)
    shutil.copytree(Path(G) / "d04" / "build_a", tampered)
    victim = tampered / "auto_pm" / "__init__.py"
    victim.write_bytes(victim.read_bytes() + b"# tampered" + NL.encode())
    # pmf_build.py verify without a deployment manifest checks only completeness (the hash check against a manifest is D07's); the D04 identity is the two-build comparison,
    # so the control is that the comparison catches one changed byte
    reference = {q.relative_to(Path(G) / "d04" / "build_b").as_posix(): sha_file(str(q)) for q in sorted((Path(G) / "d04" / "build_b").rglob("*")) if q.is_file() and q.name != "release_meta.json"}
    changed = {q.relative_to(tampered).as_posix(): sha_file(str(q)) for q in sorted(tampered.rglob("*")) if q.is_file() and q.name != "release_meta.json"}
    diff_files = sorted(k for k in set(reference) | set(changed) if reference.get(k) != changed.get(k))
    write("G8_tampered_compare.txt", json.dumps(diff_files, ensure_ascii=False) + NL)
    mutations.append({"id": "B1-a-tampered-build-is-caught-by-the-two-build-comparison", "file": "d04/build_a_tampered", "killed": diff_files == ["auto_pm/__init__.py"], "must_fail": ["payload_identical"],
                      "failed": diff_files[:4], "restored_bytes_equal": True, "last": ""})
    shutil.rmtree(tampered, ignore_errors=True)
    write("G8_mutations.json", json.dumps(mutations, ensure_ascii=False, indent=1) + NL)
    write("G8_orient_exception.json", json.dumps(orient_cases(), ensure_ascii=False, indent=1) + NL)

    # G6: trees
    tmp = Path(tempfile.mkdtemp(prefix="plugin_base_"))
    try:
        import io as _io
        import tarfile
        archive = subprocess.run(["git", "-C", str(ROOT), "archive", BASE, f"{P}/auto_pm", f"{P}/templates", f"{P}/tests", f"{P}/scripts"], capture_output=True)
        if archive.returncode == 0:
            with tarfile.open(fileobj=_io.BytesIO(archive.stdout)) as tar:
                try:
                    tar.extractall(str(tmp), filter="data")
                except TypeError:
                    tar.extractall(str(tmp))
        diffs = {}
        for sub in ("auto_pm", "templates", "tests", "scripts"):
            left, right = tree_digest(tmp / P / sub), tree_digest(Path(PROJ) / sub)
            diffs[sub] = sorted(k for k in set(left) | set(right) if left.get(k) != right.get(k))
        write("G6_tree_diff.json", json.dumps({"archive_exit": archive.returncode, "differences": diffs}, ensure_ascii=False, indent=1) + NL)
    finally:
        shutil.rmtree(tmp, ignore_errors=True)
    # G7: real pre-commit hook on a temporary index
    abs_wt = str(Path(WT).resolve())
    idx = Path(tempfile.mkdtemp(prefix="plugin_idx_")) / "index"
    env = {**ENV, "GIT_INDEX_FILE": str(idx)}
    rt = git("read-tree", "HEAD", env=env)
    ad = git("add", "--", *sorted(FILES), env=env)
    staged = git("diff", "--cached", "--name-only", env=env).stdout.splitlines()
    hook = git("rev-parse", "--git-path", "hooks/pre-commit").stdout.strip()
    hd = sh([GIT_BASH, hook], cwd=abs_wt, env=env, timeout=1800)
    write("G7_hook_dry_run.json", json.dumps({"read_tree_exit": rt.returncode, "add_exit": ad.returncode, "staged": staged, "hook_exit": hd.returncode, "tail": (hd.stdout + hd.stderr)[-400:]}, ensure_ascii=False, indent=1) + NL)
    shutil.rmtree(idx.parent, ignore_errors=True)
    shutil.rmtree(SCRATCH, ignore_errors=True)
    write("real_hooks_after.txt", hooks_record())
    write("python_procs_after.txt", py_procs() + NL)
    write("bytes_after_gates.txt", "".join(f"{sha_file(f'{WT}/{r}')} *{r}\n" for r in sorted(FILES)))
    write("final_status_after.txt", git("status", "--short", "--untracked-files=all").stdout)
    write("FAST_DONE", "done\n")


def stage_judge() -> int:
    verdict: dict = {"schema": "m2-card-gate-verdict.v1", "card": LABEL, "base": BASE, "run_id": RUN_ID, "gates": {}, "exceptions": [], "reports": {}, "deviations": []}

    def done(g: str, ok: bool, lines: list[str]) -> None:
        path = f"{G}/{g}_report.txt"
        open(path, "w", encoding="utf-8", newline="\n").write("\n".join(lines + [f"{g} VERDICT: {'PASS' if ok else 'FAIL'}"]) + "\n")
        verdict["gates"][g] = {"verdict": "PASS" if ok else "FAIL", "report": path}
        verdict["reports"][path] = sha_file(path)

    names = ["baseline_identity.json", "plugin_identity.json", "plugin_identity_after.json", "bytes_at_gates.txt", "bytes_after_gates.txt", "real_hooks_before.txt", "real_hooks_after.txt", "python_procs_before.txt",
             "python_procs_after.txt", "final_head.txt", "final_status_before.txt", "final_status_after.txt", "status_baseline.txt", "G1_baseline_candidate.txt", "G1_baseline_candidate.exit", "G1_candidate.txt",
             "G1_candidate.exit", "G3_real_plugin_tests.txt", "G3_real_plugin_tests.exit", "G3_import_probe.txt", "G3_import_probe.exit", "G3_docs.json", "G3_doc_check_baseline.txt", "G3_doc_check_baseline.exit", "G3_doc_check_candidate.txt", "G3_doc_check_candidate.exit",
             "G4_ruff.txt", "G4_ruff.exit", "G4_mypy.txt", "G4_mypy.exit", "G4_diffcheck.txt", "G4_diffcheck.exit", "G6_tree_diff.json", "G7_hook_dry_run.json", "G8_mutations.json", "install_output.json"] \
        + [f"g2_candidate/{n}.{e}" for n in GROUPS for e in ("txt", "exit")] + [f"g2_candidate/transient_{i}.{e}" for i in (1, 2, 3) for e in ("txt", "exit")]
    for n in names:
        if os.path.exists(f"{G}/{n}"):
            verdict["reports"][f"{G}/{n}"] = sha_file(f"{G}/{n}")
    rec = lambda n: {l.split(" *", 1)[1].strip(): l.split(" ", 1)[0] for l in rd(f"{G}/{n}").splitlines() if " *" in l}  # noqa: E731
    want = {r: m["sha256"] for r, m in FILES.items()}
    same = rec("bytes_at_gates.txt") == rec("bytes_after_gates.txt") == want

    ident = json.load(open(G + "/baseline_identity.json", encoding="utf-8"))
    b1, c1 = summary(G + "/G1_baseline_candidate.txt"), summary(G + "/G1_candidate.txt")
    done("G1", ident["head"] == BASE and not ident["status"].strip() and all(s["ok"] for s in ident["files"].values()) and b1["failed_ids"] == [] and c1["failed_ids"] == []
         and b1["exit"] == c1["exit"] == 0 and b1["passed"] == c1["passed"] and same,
         ["# G1 - baseline identity (HEAD is the expected start commit, clean worktree, the new report absent) and the fixture-health, smoke and import tests pass with the same counts",
          f"HEAD {ident['head'][:12]} expected {BASE[:12]}", f"baseline: {b1['last']}", f"candidate: {c1['last']}"])

    g2_lines, g2_ok = [], same
    for name in GROUPS:
        base, cand = summary(f"{CFG['baseline_g2']}/{name}.txt"), summary(f"{G}/g2_candidate/{name}.txt")
        delta = DELTAS.get(name, 0)
        if name == "rest":
            strict = cand["failed_ids"] == [] and cand["passed"] == base["passed"] + delta and cand["skipped"] == base["skipped"] + SKIPS.get(name, 0) and not cand["error_nodes"] and cand["exit"] == 0
            reruns = [f"{G}/g2_candidate/transient_{i}.txt" for i in (1, 2, 3)]
            rerun_ok = all(os.path.exists(r) and os.path.exists(r[:-4] + ".exit") for r in reruns) and all((s := summary(r))["exit"] == 0 and s["passed"] == 1 and s["failed"] == 0 for r in reruns)
            transient = cand["failed_ids"] == [R09] and cand["passed"] == base["passed"] + delta - 1 and cand["skipped"] == base["skipped"] + SKIPS.get(name, 0) and not cand["error_nodes"] and rerun_ok
            ok = strict or transient
            if transient and not strict:
                verdict["deviations"].append({"gate": "G2", "rule": "named transient-race test", "node": R09, "isolated_reruns": "3 of 3 passed in the same candidate bytes"})
            g2_lines.append(f"{name}: baseline {base['passed']}p/{base['failed']}f/{base['skipped']}s | candidate {cand['passed']}p/{cand['failed']}f/{cand['skipped']}s | expected +{delta} | strict {strict} | named transient rule {transient} -> {'OK' if ok else 'FAIL'}")
        else:
            ok = (base["failed_ids"] == cand["failed_ids"] and cand["passed"] == base["passed"] + delta and cand["skipped"] == base["skipped"] + SKIPS.get(name, 0) and not cand["error_nodes"] and cand["exit"] == base["exit"]
                  and base["deselected"] == cand["deselected"] and cand["passed"] > 0)
            g2_lines.append(f"{name}: baseline {base['passed']}p/{base['failed']}f/{base['skipped']}s | candidate {cand['passed']}p/{cand['failed']}f/{cand['skipped']}s | expected +{delta} -> {'OK' if ok else 'FAIL'}")
        g2_ok = g2_ok and ok
    done("G2", g2_ok, ["# G2 - the nine regression groups (ten outputs) against the governed outputs of the previous card (RUN-SW008-M3-REL6-002, card 218 V1.1, whose product equals the active release 1.3.3-6a9f127): every group has no failure; the counts equal the baseline plus the deltas declared in the card (measured in the V1.0 formal run RUN-SW008-M3-REL7-001, whose ten outputs are reused byte for byte; this proves zero failures and unchanged other counts, not that no test was lost)", *g2_lines])

    real = summary(G + "/G3_real_plugin_tests.txt")
    docs = json.load(open(G + "/G3_docs.json", encoding="utf-8"))
    docs_ok = all(d["ok"] for d in docs.values()) and sorted(docs) == sorted(CFG["docs"])
    doc_exit_b, doc_exit_c = (int(rd(G + f"/G3_doc_check_{t}.exit").strip()) for t in ("baseline", "candidate"))
    pi, pa = json.load(open(G + "/plugin_identity.json", encoding="utf-8")), json.load(open(G + "/plugin_identity_after.json", encoding="utf-8"))
    probe_exit = int(rd(G + "/G3_import_probe.exit").strip())
    probe_txt = rd(G + "/G3_import_probe.txt")
    f3 = json.load(open(G + "/G3_d04_d05.json", encoding="utf-8"))
    d045_ok = (f3.get("build_a_exit") == 0 and f3.get("build_b_exit") == 0 and f3.get("verify_a_exit") == 0 and f3.get("verify_b_exit") == 0 and f3.get("payload_identical") is True
               and f3.get("meta_identical_except_builder") is True and str(f3.get("meta_source_commit", "")).startswith(f3["frozen"][:7]) and orient_ok(f3.get("orient_conflicts"), (f3.get("orient_extract") or {}).get("run_id"), (f3.get("orient_extract") or {}).get("work_id"), (f3.get("orient_extract") or {}).get("work_state"), CFG.get("card_id"), f3.get("ledger_expired_running_runs_informational"))
               and all(v["exit"] == 0 and v["counts"] == [0, 0, 0] for v in f3["ledgers"].values()) and sorted(f3["ledgers"]) == sorted(CFG["projects"]))
    ex_ = f3.get("orient_extract") or {}
    orient_line = ("orient exception: conflicts=" + json.dumps(f3.get("orient_conflicts"), ensure_ascii=False) + " orient_run_id=" + str(ex_.get("run_id")) + " work_id=" + str(ex_.get("work_id"))
                   + " work_state=" + str(ex_.get("work_state")) + " ledger_expired_running_runs(informational)=" + json.dumps(f3.get("ledger_expired_running_runs_informational"), ensure_ascii=False)
                   + " verdict=" + str(orient_ok(f3.get("orient_conflicts"), ex_.get("run_id"), ex_.get("work_id"), ex_.get("work_state"), CFG.get("card_id"))))
    done("G3", real["passed"] > 0 and real["failed"] == 0 and real["skipped"] == 0 and real["exit"] == 0 and d045_ok
         and probe_exit == 0
         and docs_ok and doc_exit_c == doc_exit_b and pi["sha256"] == pa["sha256"] and same,
         ["# G3 - D04: the frozen commit is built twice, the two payloads are byte-identical and their release_meta.json equal apart from the builder, both pass verify, the payload imports from "
          "itself and its CLI answers; D05 (non-regression part): the build entry's tests pass, every registered project's ledger reconciles 0/0/0, the control root orients without conflict, the "
          "full-repository ruff/mypy audit is recorded raw; the doc check exits as at the baseline; the committed report meets its frozen line contract; the plugin binary is unchanged",
          orient_line, f"d04/d05 facts: {json.dumps({k: v for k, v in f3.items() if k != 'ledgers'}, ensure_ascii=False)[:600]}", f"ledgers: {json.dumps(f3['ledgers'], ensure_ascii=False)}",
          f"docs: {json.dumps({p.rsplit('/', 1)[-1]: d['ok'] for p, d in docs.items()}, ensure_ascii=False)}; doc check exit baseline {doc_exit_b} candidate {doc_exit_c}",
          f"tests: {real['last']}", f"import probe: exit {probe_exit} {probe_txt.strip()[:80]}"])

    ruff_exit, mypy_exit, dc_exit = (int(rd(G + f"/{n}.exit").strip()) for n in ("G4_ruff", "G4_mypy", "G4_diffcheck"))
    mypy_txt = rd(G + "/G4_mypy.txt")
    done("G4", ruff_exit == 0 and mypy_exit == 0 and "Success: no issues found" in mypy_txt and dc_exit == 0 and same,
         ["# G4 - `ruff check auto_pm tests` clean, `mypy --platform win32 auto_pm` clean, git diff --check", f"ruff exit {ruff_exit}; mypy exit {mypy_exit}: {mypy_txt.strip().splitlines()[-1] if mypy_txt.strip() else ''}; diff --check exit {dc_exit}"])

    status_after = [l[3:].strip('"') for l in rd(G + "/final_status_after.txt").splitlines() if l.strip()]
    head = rd(G + "/final_head.txt").strip()
    actual = {r: sha_file(f"{WT}/{r}") for r in FILES}
    residue = [p for p in status_after if re.search(r"\.coverage|/tmp_|__pycache__|\.pytest_cache|htmlcov", p)]
    done("G5", sorted(status_after) == sorted(FILES) and actual == want and head == BASE and not rd(G + "/status_baseline.txt").strip() and not residue and same,
         [f"# G5 - exactly the {len(FILES)} frozen paths with the frozen bytes; HEAD is the baseline; the worktree was clean before the install", f"changed: {len(status_after)} paths", f"HEAD == baseline: {head == BASE}", f"residue: {residue}"])

    td = json.load(open(G + "/G6_tree_diff.json", encoding="utf-8"))
    expect: dict[str, list[str]] = {sub: [] for sub in ("auto_pm", "templates", "tests", "scripts")}
    for r in FILES:
        if not r.startswith(P + "/"):
            continue
        sub, _, tail = rel(r).partition("/")
        if sub in expect:
            expect[sub].append(tail)
    done("G6", td["archive_exit"] == 0 and all(sorted(td["differences"][k]) == sorted(expect[k]) for k in expect),
         ["# G6 - each of auto_pm, templates, tests and scripts differs from the baseline exactly by the frozen files (also after the mutations were restored)", json.dumps({k: len(v) for k, v in td["differences"].items()})])

    hk = json.load(open(G + "/G7_hook_dry_run.json", encoding="utf-8"))
    hooks_same = rd(G + "/real_hooks_before.txt") == rd(G + "/real_hooks_after.txt") and bool(rd(G + "/real_hooks_before.txt").strip())
    done("G7", hooks_same and rd(G + "/python_procs_before.txt") == rd(G + "/python_procs_after.txt") and hk["hook_exit"] == 0 and sorted(hk["staged"]) == sorted(FILES) and same,
         [f"# G7 - real hooks unchanged, no python left, the real pre-commit hook accepts exactly the {len(FILES)} staged files (temporary index)", f"hook exit {hk['hook_exit']}"])

    muts = json.load(open(G + "/G8_mutations.json", encoding="utf-8"))
    oc_ = json.load(open(G + "/G8_orient_exception.json", encoding="utf-8"))
    done("G8", len(muts) == 3 and all(m["killed"] and m["restored_bytes_equal"] for m in muts) and same and len(oc_) == len(orient_cases()) and len(oc_) >= 26 and all(c["got"] == c["expected"] for c in oc_),
         ["# G8 - a tampered copy of the build (one byte added to auto_pm/__init__.py) is caught by the two-build comparison; two in-memory document controls (the report losing its proof sentence, the report losing the full frozen commit) are refused "
          "by the line contract",
          *[f"{m['id']}: killed={m['killed']} must_fail={m['must_fail']} failed={m['failed'][:4]} restored={m['restored_bytes_equal']}" for m in muts]])

    done("G9", same and sorted(status_after) == sorted(FILES) and head == BASE, ["# G9 - bytes and change set after all gates are those before them", f"{len(status_after)} paths"])
    json.dump(verdict, open(f"{G}/gate_verdict.json", "w", encoding="utf-8"), ensure_ascii=False, indent=1)
    print({g: d["verdict"] for g, d in verdict["gates"].items()})
    return 0 if all(d["verdict"] == "PASS" for d in verdict["gates"].values()) and len(verdict["gates"]) == 9 else 1


if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("--stage", choices=["baseline", "fast", "judge"], required=True)
    st = ap.parse_args().stage
    sys.exit(stage_baseline() if st == "baseline" else stage_judge() if st == "judge" else (stage_fast() or 0))
